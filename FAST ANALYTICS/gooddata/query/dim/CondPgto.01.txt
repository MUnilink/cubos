SELECT 'P |01|SE4010|'+ COALESCE(NULLIF(RTRIM(COALESCE(E4_FILIAL, ' '))+'|'+RTRIM(COALESCE(E4_CODIGO, ' ')), ' '), '|') AS BK_CONDICAO_DE_PAGAMENTO,
       SE4.E4_CODIGO AS COD_CONDICAO_PAGAMENTO,
       SE4.E4_DESCRI AS DESC_CONDICAO_PAGAMENTO
FROM SE4010 SE4
WHERE SE4.D_E_L_E_T_ = ' '
UNION
SELECT 'P |01|SE4010||',
       '01 - INDEFINIDO',
       '01 - INDEFINIDO'

SELECT 'P |01|SA1010|'+ COALESCE(NULLIF(RTRIM(COALESCE(A1_FILIAL, ' '))+'|'+RTRIM(COALESCE(A1_COD, ' '))+RTRIM(COALESCE(A1_LOJA, ' ')), ' '), '|') AS BK_CLIENTE,
       SA1.A1_COD AS COD_CLIENTE,
       SA1.A1_LOJA AS LOJA_CLIENTE,
       trim(SA1.A1_NOME) AS NOME_CLIENTE,
       trim(SX5.X5_DESCRI) AS SEGMENTACAO_ATIVIDADE_1
FROM SA1010 SA1
LEFT JOIN SX5010 SX5 ON SA1.A1_SATIV1 = X5_CHAVE
AND X5_TABELA='T3'
WHERE SA1.D_E_L_E_T_ = ' '
  AND SA1.A1_SATIV1 <>'999999'
UNION
SELECT 'P |01|SA1010||',
       '01 - INDEFINIDO',
       '01 - INDEFINIDO',
       '01 - INDEFINIDO',
       '01 - INDEFINIDO'

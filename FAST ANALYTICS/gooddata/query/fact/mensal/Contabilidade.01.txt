SELECT BK_EMPRESA,BK_FILIAL,BK_CONTA,NORMAL,CTARES,TIPOCONTA,GRUPO,NATCTA,CT1DTEXSF,DESCCTA,DATA,CAST( COALESCE(SALDOANTDB,0) AS DECIMAL(16,2)) AS SALDOANTDB,CAST( COALESCE(SALDOANTCR,0) AS DECIMAL(16,2)) AS SALDOANTCR,CAST( COALESCE(SALDODEB,0) AS DECIMAL(16,2)) AS SALDODEB,CAST( COALESCE(SALDOCRD,0) AS DECIMAL(16,2)) AS SALDOCRD FROM (SELECT DISTINCT 'P |01|01' AS BK_EMPRESA,CASE WHEN CQ1_FILIAL IS NULL THEN 'P |01||' ELSE 'P |01|01'+ CAST(CQ1_FILIAL AS CHAR (6) ) END AS BK_FILIAL,'P |01|CT1010|'+  COALESCE(NULLIF(RTRIM( COALESCE(CT1_FILIAL,' ') )+'|'+RTRIM( COALESCE(CQ1_CONTA,' ') ),' '),'|') AS BK_CONTA,CT1_NORMAL AS NORMAL,CT1_RES AS CTARES,CT1_CLASSE AS TIPOCONTA,CT1_GRUPO AS GRUPO,CT1_NATCTA AS NATCTA,CT1_DTEXSF AS CT1DTEXSF,CT1_DESC01 AS DESCCTA,<<FINAL_DATE>> AS DATA,(SELECT CASE WHEN SUM(CQ1_DEBITO) IS NULL THEN 0 ELSE SUM(CQ1_DEBITO) END AS SALDOANTDB FROM CQ1010 CQ11 WHERE  CQ11.CQ1_CONTA = CQ1.CQ1_CONTA AND CQ11.CQ1_TPSALD = '1' AND CQ11.CQ1_DATA < <<START_DATE>> AND CQ11.D_E_L_E_T_ = ' ' AND CQ11.CQ1_FILIAL = CQ1.CQ1_FILIAL AND CQ11.CQ1_MOEDA = '01')  AS SALDOANTDB,(SELECT CASE WHEN SUM(CQ1_CREDIT) IS NULL THEN 0 ELSE SUM(CQ1_CREDIT) END AS SALDOANTCR FROM CQ1010 CQ12 WHERE  CQ12.CQ1_CONTA = CQ1.CQ1_CONTA AND CQ12.CQ1_TPSALD = '1' AND CQ12.CQ1_DATA < <<START_DATE>> AND CQ12.D_E_L_E_T_ = ' ' AND CQ12.CQ1_FILIAL = CQ1.CQ1_FILIAL AND CQ12.CQ1_MOEDA = '01')  AS SALDOANTCR,(SELECT CASE WHEN SUM(CQ1_DEBITO) IS NULL THEN 0 ELSE SUM(CQ1_DEBITO) END AS SALDODEB FROM CQ1010 CQ13 WHERE  CQ13.CQ1_CONTA = CQ1.CQ1_CONTA AND CQ13.CQ1_TPSALD = '1' AND CQ13.CQ1_DATA >= <<START_DATE>> AND CQ13.CQ1_DATA <= <<FINAL_DATE>> AND CQ13.D_E_L_E_T_ = ' ' AND CQ13.CQ1_FILIAL = CQ1.CQ1_FILIAL AND CQ13.CQ1_MOEDA = '01')  AS SALDODEB,(SELECT CASE WHEN SUM(CQ1_CREDIT) IS NULL THEN 0 ELSE SUM(CQ1_CREDIT) END AS SALDOCRD FROM CQ1010 CQ14 WHERE  CQ14.CQ1_CONTA = CQ1.CQ1_CONTA AND CQ14.CQ1_TPSALD = '1' AND CQ14.CQ1_DATA >= <<START_DATE>> AND CQ14.CQ1_DATA <= <<FINAL_DATE>> AND CQ14.D_E_L_E_T_ = ' ' AND CQ14.CQ1_FILIAL = CQ1.CQ1_FILIAL AND CQ14.CQ1_MOEDA = '01')  AS SALDOCRD FROM CQ1010 CQ1 INNER JOIN CT1010 CT1 ON CT1_FILIAL = CQ1_FILIAL AND CT1_CONTA = CQ1_CONTA AND CT1.D_E_L_E_T_ = ' ' WHERE  CQ1.D_E_L_E_T_ = ' ' AND CQ1.CQ1_MOEDA = '01' )  SALDOARQ WHERE  (SALDOANTDB - SALDOANTCR <> 0 OR SALDODEB <> 0 OR SALDOCRD <> 0)

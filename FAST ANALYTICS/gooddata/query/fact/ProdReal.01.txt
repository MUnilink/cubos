SELECT 'P |01|01' AS BK_EMPRESA,
       CASE
           WHEN D3_FILIAL IS NULL THEN 'P |01||'
           ELSE 'P |01|01'+ CAST(D3_FILIAL AS CHAR (6))
       END AS BK_FILIAL,
       CASE
           WHEN
                  (SELECT COUNT(*)
                   FROM SG1010 SG1
                   WHERE G1_FILIAL = C2_FILIAL
                     AND G1_COD = C2_PRODUTO) = 0 THEN 'Comprado'
           WHEN (
                   (SELECT COUNT(*)
                    FROM SG1010 SG1
                    WHERE G1_FILIAL = C2_FILIAL
                      AND G1_COD = C2_PRODUTO) > 0)
                AND (B1_TIPO LIKE '%MP%') THEN 'Mat�ria-Prima'
           WHEN (
                   (SELECT COUNT(*)
                    FROM SG1010 SG1
                    WHERE G1_FILIAL = C2_FILIAL
                      AND G1_COD = C2_PRODUTO) > 0)
                AND (B1_TIPO NOT LIKE '%MP%') THEN 'Fabricado'
       END AS TIPO_PRODUTO,
       CASE
           WHEN C2_TPOP = 'F' THEN 'Firme'
           ELSE 'Prevista'
       END AS TIPO_PRODUCAO,
       C2_PEDIDO AS PEDIDO,
       C2_SEQUEN AS SEQUENCIA_ORDEM,
       C2_PRIOR AS PRIORIDADE_ORDEM,
       C2_NUM+C2_ITEM+C2_SEQUEN+C2_ITEMGRD AS ORDEM,<<EXTRACTION_DATE>> AS DATA_EXTRACAO,
                                                      D3_EMISSAO AS DATA_EMISSAO_ORDEM,
                                                      C2_DATRF AS DATA_TERMINO_ORDEM,
                                                      'P |01|SB1010|'+ COALESCE(NULLIF(RTRIM(COALESCE(B1_FILIAL, ' '))+'|'+RTRIM(COALESCE(B1_COD, ' ')), ' '), '|') AS BK_ITEM,
                                                      'P |01|SX5010|'+ COALESCE(NULLIF(RTRIM(COALESCE(B1_FILIAL, ' '))+'|'+RTRIM(COALESCE(B1_TIPO, ' ')), ' '), '|') AS BK_FAMILIA_MATERIAL,
                                                      ' ' AS BK_FAMILIA_COMERCIAL,
                                                      'P |01|SBM010|'+ COALESCE(NULLIF(RTRIM(COALESCE(B1_FILIAL, ' '))+'|'+RTRIM(COALESCE(B1_GRUPO, ' ')), ' '), '|') AS BK_GRUPO_ESTOQUE,
                                                      'P |01|SA1010|'+ COALESCE(NULLIF(RTRIM(COALESCE(A1_FILIAL, ' '))+'|'+RTRIM(COALESCE(A1_TIPO, ' ')), ' '), '|') AS BK_GRUPO_CLIENTE,
                                                      'P |01|SB2010|'+ COALESCE(NULLIF(RTRIM(COALESCE(C2_FILIAL, ' '))+'|'+RTRIM(COALESCE(C2_LOCAL, ' ')), ' '), '|') AS BK_DEPOSITO,
                                                      'P |01|SA1010|'+ COALESCE(NULLIF(RTRIM(COALESCE(B1_FILIAL, ' '))+'|'+RTRIM(COALESCE(A1_COD, ' '))+RTRIM(COALESCE(A1_LOJA, ' ')), ' '), '|') AS BK_CLIENTE,
                                                      CASE
                                                          WHEN A1_COD_MUN = ' ' THEN 'P |01|CC2010|'+ COALESCE(NULLIF(RTRIM(COALESCE(A1_EST, ' ')), ' '), '|')
                                                          ELSE 'P |01|CC2010|'+ COALESCE(NULLIF(RTRIM(COALESCE(A1_EST, ' '))+RTRIM(COALESCE(A1_COD_MUN, ' ')), ' '), '|')
                                                      END AS BK_REGIAO,
                                                      D3_QUANT AS QTD_PRODUCAO,
                                                      D3_PERDA AS QTD_REFUGADA,
                                                      D3_CUSTO1 AS VAL_CUST_PRODUCAO,
                                                      (C2_QUANT * B2_CM1) AS VAL_PROGRAMADO,

  (SELECT COALESCE(SUM(SD3B.D3_CUSTO1 / (CASE
                                             WHEN SB2B.B2_CM1 = 0 THEN 1
                                             ELSE SB2B.B2_CM1
                                         END)), 0)
   FROM SD3010 SD3B
   LEFT JOIN SB2010 SB2B ON SB2B.B2_FILIAL = D3_FILIAL
   AND SB2B.B2_COD = SD3B.D3_COD
   AND SB2B.B2_LOCAL = SD3B.D3_LOCAL
   WHERE SD3B.D3_OP = SD3.D3_OP
     AND SD3B.D3_ESTORNO <> 'S'
     AND SD3B.D3_CF IN ('RE0',
                        'RE1')
     AND (SUBSTRING(SD3B.D3_COD, 1, 3) <> 'MOD'
          AND
            (SELECT SB1B.B1_CCCUSTO
             FROM SB1010 SB1B
             WHERE SB1B.B1_FILIAL = '      '
               AND SB1B.B1_COD = SD3B.D3_COD
               AND SB1B.D_E_L_E_T_ = ' ') = ' ')) AS VAL_MATERIAL_PROGRAMADO,

  (SELECT COALESCE(SUM(SD3B.D3_CUSTO1 / (CASE
                                             WHEN SB2B.B2_CM1 = 0 THEN 1
                                             ELSE SB2B.B2_CM1
                                         END)), 0)
   FROM SD3010 SD3B
   LEFT JOIN SB2010 SB2B ON SB2B.B2_FILIAL = D3_FILIAL
   AND SB2B.B2_COD = SD3B.D3_COD
   AND SB2B.B2_LOCAL = SD3B.D3_LOCAL
   WHERE SD3B.D3_OP = SD3.D3_OP
     AND SD3B.D3_ESTORNO <> 'S'
     AND SD3B.D3_CF IN ('RE0',
                        'RE1')
     AND (SUBSTRING(SD3B.D3_COD, 1, 3) = 'MOD'
          OR
            (SELECT SB1B.B1_CCCUSTO
             FROM SB1010 SB1B
             WHERE SB1B.B1_FILIAL = '      '
               AND SB1B.B1_COD = SD3B.D3_COD
               AND SB1B.D_E_L_E_T_ = ' ') <> ' ')) AS VAL_MAO_DE_OBRA_PROGRAMADO,
                                                      CASE
                                                          WHEN D3_PERDA > 0 THEN ((D3_CUSTO1 / (D3_QUANT + D3_PERDA)) * D3_PERDA)
                                                      END AS VAL_CUSTO_REFUGADO,
                                                      C2_EMISSAO AS INICIO_MOVIMENTO_REAL,
                                                      C2_DATRF AS FIM_MOVIMENTO_REAL,
                                                      C2_DATPRI AS INICIO_MOVIMENTO_PREVISTO,
                                                      C2_DATPRF AS FIM_MOVIMENTO_PREVISTO,
                                                      CASE
                                                          WHEN D3_QUANT > 0 THEN ROUND(D3_CUSTO1 / D3_QUANT, 2)
                                                      END AS VL_UNITARIO_PRODUTO,
                                                      D3_EMISSAO AS DATA
FROM SD3010 SD3
INNER JOIN SC2010 SC2 ON C2_FILIAL = D3_FILIAL
AND C2_NUM+C2_ITEM+C2_SEQUEN+C2_ITEMGRD = D3_OP
AND SC2.D_E_L_E_T_= ' '
LEFT JOIN SC5010 SC5 ON C5_FILIAL = C2_FILIAL
AND C5_NUM = SC2.C2_PEDIDO
LEFT JOIN SA1010 SA1 ON A1_FILIAL = '      '
AND A1_COD = SC5.C5_CLIENTE
AND A1_LOJA = C5_LOJACLI
INNER JOIN SB1010 SB1 ON B1_FILIAL = '      '
AND B1_COD = D3_COD
AND SB1.D_E_L_E_T_ = ' '
LEFT JOIN SB2010 SB2 ON B2_FILIAL = C2_FILIAL
AND B2_COD = C2_PRODUTO
AND B2_LOCAL = SB1.B1_LOCPAD
WHERE D3_CF IN ('PR0',
                'PR1')
  AND SD3.D_E_L_E_T_ = ' '
  AND SD3.D3_ESTORNO <> 'S'
  AND SD3.D3_EMISSAO BETWEEN <<START_DATE>> AND <<FINAL_DATE>>
